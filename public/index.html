Quilon

A statically-typed, symbol-based language that compiles to native and should make me laugh.

Colorless concurrency on cooperative fibers — work runs when its data is ready, wherever it sits in the file.

~ ` holes splice any expression in, and a
~ record can override how it renders
User = {
  name :: Text,
  age :: Num,
  ` = () -> Text => < "`it.name` (`it.age`)" >
}

^ = () -> $ => <
  io.print("half is `1 / 2`")   ~ half is 0.5
  io.print("sum `1 + 2 + 3`")   ~ sum 6

  u = User { name = "Ada", age = 36 }
  io.print("hi `u`!")           ~ hi Ada (36)!
>

~ lo <- hi is an inclusive range; fed straight
~ to an array method it counts in place
sumTo = (n :: Num) -> Num => <
  (1 <- n).reduce(0, (acc, x) => acc + x)
>

^ = () -> $ => <
  (1 <- 3).each(i => io.print("tick `i`"))
  io.print(sumTo(100))         ~ 5050
  io.print(sumTo(100000000))   ~ 5000000050000000
>
<< core.io

~ records carry methods; `it` is the instance.
~ := declares a setter; = methods are verified
~ non-mutating
Counter = {
  value :: Num,

  bump := (by :: Num) => < it.value := it.value + by >,
  peek = () -> Num => < it.value >
}

^ = () -> $ => <
  c := Counter { value = 30 }
  c.bump(5)
  io.print(c.peek())   ~ 35
>
<< core.io

~ the binding operator IS the capture mode,
~ and a function may hand a closure back
mkCounter = () -> () -> Num => <
  count := 0                ~ := captures by reference
  () -> Num => <
    count := count + 1
    count
  >
>

adder = (n :: Num) -> (Num) -> Num => < (x) => x + n >

^ = () -> $ => <
  tick = mkCounter()
  tick()
  io.print(tick())          ~ 2
  add5 = adder(5)           ~ = captures by value
  io.print(add5(37))        ~ 42
>

~ variants split by /, payloads optional
Shape = Circle(Num) / Rect(Num, Num)

~ ? and | match, and bind the payload
area = (s :: Shape) -> Num => < s ?
  | Circle(r)  => 3 * r * r
  | Rect(w, h) => w * h
>

~ Result is a normal sum type: Ok / NotOk
check = (n :: Num) -> Result => <
  n <= 100 ? Ok($) : NotOk(n)
>

^ = () -> $ => <
  io.print(area(Rect(6, 7)))   ~ 42
  io.print(check(50) ?
    | Ok(_)    => 0
    | NotOk(c) => c)
>

~ @ marks the leaf IO primitive alone; every
~ function that calls it stays plain code,
~ the same shape as any other
pause = (s :: Num) -> $ => < @sleep(s) >

^ = () -> $ => <
  start = time.now()
  ~ parks the fiber; the thread keeps working
  pause(0.05)
  io.print("`time.now() - start >= 0.05`")   ~ True
>

~ Request.get builds it, send() answers Ok / NotOk:
~ a network failure is a value to match, not a crash
show = (page :: http.Response) -> $ => <
  io.print(page.status())                        ~ 200
  io.print(page.headers().get("Content-Type") ?
    | Ok(t)    => t                              ~ text/html
    | NotOk(_) => "no content type")
  io.print(page.body().contains("Example Domain"))   ~ True
>

^ = () -> $ => <
  http.Request.get("http://example.com/").send() ?
    | Ok(page)   => show(page)
    | NotOk(err) => io.eprint("no luck: `err`")
>

~ tests sit beside the code they check: `quilon test`
~ runs the blocks, every other command erases them
double = (n :: Num) -> Num => < n * 2 >

test.describe("double", () => <
  test.it("doubles", () => expect(double(21), equals(42)))
  test.it("keeps zero", () => expect(double(0), equals(0)))
>)

^ = () -> $ => < io.print(double(4)) >   ~ 8
